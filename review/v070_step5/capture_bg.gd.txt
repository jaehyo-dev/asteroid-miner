extends SceneTree
## 시안 뒤에 깔 진짜 출격 화면: 같은 시드로 약 80초 플레이한 상태
var run: Node
var frame := 0
func _process(_d: float) -> bool:
	frame += 1
	if frame == 1:
		var a := OS.get_cmdline_user_args()
		root.size = Vector2i(int(a[0]), int(a[1]))
		root.get_node("App").fit_screen()
		run = load("res://scenes/run.tscn").instantiate()
		root.add_child(run)
	elif frame == 3:
		run.sim.rng.seed = 4242
		run.sim.max_hp = 1.0e6
		run.sim.hp = 1.0e6
		var move := Vector2.ZERO
		while run.sim.time < 80.0:
			if not run.sim.choices.is_empty():
				run.sim.choose(0)
				continue
			var danger = RunSim.nearest_enemy(run.sim.player_pos, run.sim.enemies, 70.0)
			if danger != null:
				move = (run.sim.player_pos - danger.pos).normalized()
			else:
				move = Vector2.ZERO
				var best := 260.0
				for m in run.sim.minerals:
					var d: float = m.pos.distance_to(run.sim.player_pos)
					if d < best:
						best = d
						move = (m.pos - run.sim.player_pos).normalized()
			run.sim.update(1.0 / 30.0, move)
		run.sim.apply_upgrades()
		run.sim.max_hp = Balance.SHIP_MAX_HP + 20.0
		run.sim.hp = run.sim.max_hp * 0.7
		run.sim.xp = -100000
	elif frame > 3 and run.current_menu == run.Menu.PAUSE:
		run._close_menu()
	if frame == 14:
		var a := OS.get_cmdline_user_args()
		root.get_texture().get_image().save_png(a[2])
		print("저장 ", root.size, " 논리 ", root.content_scale_size)
	return frame >= 15
